cmake_minimum_required(VERSION 4.2)
project(BonDriverRelay VERSION 0.1.0 LANGUAGES CXX)

if(NOT WIN32 OR NOT MSVC)
    message(FATAL_ERROR "BonDriverRelay requires Windows and MSVC (Visual Studio 2026).")
endif()
if(NOT CMAKE_GENERATOR STREQUAL "Visual Studio 18 2026")
    message(FATAL_ERROR "Use the Visual Studio 18 2026 generator to produce .slnx solutions.")
endif()

set(CMAKE_CONFIGURATION_TYPES "Debug;Release" CACHE STRING "Supported configurations" FORCE)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>DLL")
# Keep compiler debug symbols in each .obj instead of a shared compiler PDB.
# The linker still emits the final Debug PDB for native debugging.
set(CMAKE_MSVC_DEBUG_INFORMATION_FORMAT "$<$<CONFIG:Debug>:Embedded>")
set_property(GLOBAL PROPERTY USE_FOLDERS ON)

option(BDR_ENABLE_ANALYSIS "Enable MSVC native code analysis" OFF)
add_library(bdr_build_options INTERFACE)
target_compile_options(bdr_build_options INTERFACE /W4 /WX /permissive- /utf-8 /EHsc /GR /Gd)
target_compile_definitions(bdr_build_options INTERFACE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
if(BDR_ENABLE_ANALYSIS)
    # Analyze our sources; pinned third-party headers are SYSTEM/external.
    target_compile_options(bdr_build_options INTERFACE /analyze /analyze:external-)
endif()

include(cmake/Dependencies.cmake)
configure_file(cmake/Version.h.in generated/BonDriverRelay/Version.h @ONLY)
add_library(bdr_protocol STATIC src/protocol/Header.cpp src/protocol/Control.cpp)
target_include_directories(bdr_protocol PUBLIC include "${CMAKE_CURRENT_BINARY_DIR}/generated")
target_link_libraries(bdr_protocol PUBLIC bdr_build_options)

add_library(bdr_core STATIC src/core/Error.cpp src/core/Paths.cpp src/core/Configuration.cpp
    src/core/Logger.cpp src/core/BonDriverAdapter.cpp src/core/TunerSession.cpp src/core/ConfigurationReloader.cpp)
target_include_directories(bdr_core PUBLIC include)
target_link_libraries(bdr_core PUBLIC bdr_build_options PRIVATE spdlog::spdlog tomlplusplus::tomlplusplus)
target_link_libraries(bdr_protocol PUBLIC bdr_core)

add_library(bdr_runtime STATIC src/transport/Tcp.cpp src/core/TsDistributor.cpp src/core/TunerWorker.cpp src/core/TunerManager.cpp src/server/RelayServer.cpp)
target_link_libraries(bdr_runtime PUBLIC bdr_protocol bdr_core ws2_32)
add_library(bdr_decoder STATIC src/decoder/B25DecoderAdapter.cpp)
target_include_directories(bdr_decoder SYSTEM PRIVATE third_party/libaribb25)
target_link_libraries(bdr_decoder PUBLIC bdr_core)
target_link_libraries(bdr_runtime PUBLIC bdr_decoder)
add_library(bdr_client STATIC src/client/RemoteClient.cpp src/client/BonDriverRelayClient.cpp)
target_link_libraries(bdr_client PUBLIC bdr_runtime)

add_library(bdr_host STATIC src/service/HostRuntime.cpp src/service/WindowsServiceHost.cpp)
add_library(bdr_monitor STATIC src/monitor/Http.cpp src/monitor/MonitorView.cpp src/monitor/MonitorServer.cpp)
target_link_libraries(bdr_monitor PUBLIC bdr_runtime)
target_link_libraries(bdr_host PUBLIC bdr_monitor advapi32)
add_executable(BonDriverRelayServer src/server/main.cpp)
target_include_directories(BonDriverRelayServer PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/generated")
target_link_libraries(BonDriverRelayServer PRIVATE bdr_host)
add_library(BonDriver_Relay SHARED src/client/Factory.cpp src/client/BonDriver_Relay.def)
target_link_libraries(BonDriver_Relay PRIVATE bdr_client)
set_property(DIRECTORY PROPERTY VS_STARTUP_PROJECT BonDriverRelayServer)
add_executable(BonDriverRelayProbe src/tools/Probe.cpp)
target_link_libraries(BonDriverRelayProbe PRIVATE bdr_core)

option(BUILD_TESTING "Enable optional local checks" OFF)
if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/tests/CMakeLists.txt")
    include(CTest)
    if(BUILD_TESTING)
        add_subdirectory(tests)
    endif()
endif()

# An explicit file list prevents accidentally packaging user-provided external DLLs.
install(TARGETS BonDriverRelayServer BonDriverRelayProbe RUNTIME DESTINATION .)
install(TARGETS BonDriver_Relay RUNTIME DESTINATION client)
install(FILES config/server.release.toml DESTINATION . RENAME server.toml)
install(FILES config/client.example.toml DESTINATION client RENAME BonDriver_Relay.toml)
install(FILES tools/Start-Server.cmd tools/Check-Config.cmd DESTINATION .)
install(FILES README.md LICENSE THIRD_PARTY_NOTICES.md DESTINATION .)
install(FILES third_party/libaribb25/LICENSE third_party/libaribb25/NOTICE third_party/libaribb25/SOURCE.md DESTINATION licenses/libaribb25)
